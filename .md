```#include <iostream>
#include <iomanip>

using namespace std;

// Структура для хранения интервала времени
struct TimeInterval {
    int hours;
    int minutes;
    int seconds;
};

// Функция для перевода времени в секунды
int toSeconds(const TimeInterval& t) {
    return t.hours * 3600 + t.minutes * 60 + t.seconds;
}

// Функция для нормализации времени
TimeInterval normalizeTime(int totalSeconds) {
    TimeInterval result;
    result.hours = totalSeconds / 3600;
    result.minutes = (totalSeconds % 3600) / 60;
    result.seconds = totalSeconds % 60;
    return result;
}

// Функция для суммирования интервалов времени
TimeInterval sumIntervals(TimeInterval* intervals, int N) {
    int totalSeconds = 0;
    
    for (int i = 0; i < N; i++) {
        totalSeconds += toSeconds(intervals[i]);
    }
    
    return normalizeTime(totalSeconds);
}

// Функция для ввода всех интервалов
void inputIntervals(TimeInterval* intervals, int N) {
    cout << "\nВведите " << N << " интервалов времени:\n";
    
    for (int i = 0; i < N; i++) {
        cout << "Интервал " << i + 1 << ":\n";
        
        do {
            cout << "  Часы (0-23): ";
            cin >> intervals[i].hours;
        } while (intervals[i].hours < 0 || intervals[i].hours > 23);
        
        do {
            cout << "  Минуты (0-59): ";
            cin >> intervals[i].minutes;
        } while (intervals[i].minutes < 0 || intervals[i].minutes > 59);
        
        do {
            cout << "  Секунды (0-59): ";
            cin >> intervals[i].seconds;
        } while (intervals[i].seconds < 0 || intervals[i].seconds > 59);
        
        cout << endl;
    }
}

// Функция для вывода времени в формате HH:MM:SS
void printTime(const TimeInterval& time) {
    cout << setfill('0')
         << setw(2) << time.hours << ":"
         << setw(2) << time.minutes << ":"
         << setw(2) << time.seconds;
}

int main() {
    int N;
    
    // Ввод количества интервалов
    cout << "Система мониторинга пиковой нагрузки портала\n";
    cout << "=============================================\n";
    cout << "Введите количество интервалов N (10-30): ";
    cin >> N;
    
    if (N < 10 || N > 30) {
        cout << "Ошибка! N должно быть в диапазоне 10-30.\n";
        return 1;
    }
    
    // Создание динамического массива
    TimeInterval* intervals = new (nothrow) TimeInterval[N];
    
    if (intervals == nullptr) {
        cout << "Ошибка выделения памяти!\n";
        return 1;
    }
    
    // Ввод данных
    inputIntervals(intervals, N);
    
    // Вывод введенных данных
    cout << "\nЗафиксированные интервалы нагрузки:\n";
    for (int i = 0; i < N; i++) {
        cout << "  " << i + 1 << ". ";
        printTime(intervals[i]);
        cout << endl;
    }
    
    // Расчет и вывод результата
    TimeInterval total = sumIntervals(intervals, N);
    
    cout << "\n=============================================\n";
    cout << "ИТОГ: Суммарное время работы портала: ";
    printTime(total);
    cout << " (" << total.hours << " часов "
         << total.minutes << " минут "
         << total.seconds << " секунд)\n";
    cout << "=============================================\n";
    
    // Освобождение памяти
    delete[] intervals;
    
    return 0;
}
```
