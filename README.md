# Блок-схема программы мониторинга пиковой нагрузки портала

## Описание программы
Программа суммирует N интервалов времени (10 ≤ N ≤ 30) и вычисляет общее время работы портала.

## Диаграмма алгоритма

```mermaid
flowchart TD
    Start([Начало]) --> A[Вывод заголовка]
    A --> B[Ввод N 10-30]
    
    B --> C{10 ≤ N ≤ 30?}
    C -->|Нет| D[Ошибка: неверный N]
    D --> End1([Завершение])
    
    C -->|Да| E[Выделение памяти]
    
    E --> F{Память выделена?}
    F -->|Нет| G[Ошибка памяти]
    G --> End2([Завершение])
    
    F -->|Да| H[Ввод интервалов]
    
    subgraph H[Функция inputIntervals]
        H1[Заголовок ввода] --> H2[Цикл i=0..N-1]
        H2 --> H3[Ввод часов 0-23]
        H3 --> H4[Ввод минут 0-59]
        H4 --> H5[Ввод секунд 0-59]
        H5 --> H6[i++]
        H6 --> H7{i < N?}
        H7 -->|Да| H3
        H7 -->|Нет| H8[Конец функции]
    end
    
    H --> I[Вывод заголовка<br>интервалов]
    I --> J[Цикл j=0..N-1]
    J --> K[Вызов printTime]
    K --> L[j++]
    L --> M{j < N?}
    M -->|Да| K
    M -->|Нет| N[Суммирование времени]
    
    subgraph N[Функция sumIntervals]
        N1[totalSeconds = 0] --> N2[Цикл k=0..N-1]
        N2 --> N3[Вызов toSeconds]
        N3 --> N4[Добавить к totalSeconds]
        N4 --> N5[k++]
        N5 --> N6{k < N?}
        N6 -->|Нет| N7[Вызов normalizeTime]
        N6 -->|Да| N3
        
        subgraph N7[Функция normalizeTime]
            O1[Вычисление hours = totalSeconds / 3600]
            O2[Вычисление minutes = остаток / 60]
            O3[Вычисление seconds = остаток % 60]
            O4[Возврат TimeInterval]
        end
        
        N7 --> N8[Возврат результата]
    end
    
    subgraph K[Функция printTime]
        P1[Установка setfill'0'] --> P2[Установка ширины 2]
        P2 --> P3[Вывод часов с ведущим 0]
        P3 --> P4[Вывод двоеточия]
        P4 --> P5[Вывод минут с ведущим 0]
        P5 --> P6[Вывод двоеточия]
        P6 --> P7[Вывод секунд с ведущим 0]
        P7 --> P8[Конец функции]
    end
    
    subgraph N3[Функция toSeconds]
        Q1[Вычисление: hours*3600]
        Q2[Вычисление: minutes*60]
        Q3[Суммирование: + seconds]
        Q4[Возврат общего<br>количества секунд]
    end
    
    N --> R[Вывод разделителя]
    R --> S[Вызов printTime<br>для общего времени]
    S --> T[Вывод времени текстом]
    T --> U[Освобождение памяти]
    U --> End3([Конец программы])
```
